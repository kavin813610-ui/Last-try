import React, { useState } from 'react';
import { useStore } from '../context/StoreContext';
import {
  ShieldCheck,
  Truck,
  RotateCcw,
  CheckCircle2,
  Clock,
  Heart,
  ShoppingBag,
  Star,
  MapPin,
  ChevronRight
} from 'lucide-react';

export const ProductDetailPage: React.FC = () => {
  const {
    selectedProduct,
    addToCart,
    navigate,
    toggleWishlist,
    isInWishlist
  } = useStore();

  if (!selectedProduct) {
    return (
      <div className="max-w-4xl mx-auto px-4 py-20 text-center">
        <h2 className="font-serif-display text-2xl font-bold">No Timepiece Selected</h2>
        <button
          onClick={() => navigate('shop')}
          className="mt-4 bg-neutral-950 text-white px-6 py-2 rounded-lg text-sm"
        >
          Return to Shop
        </button>
      </div>
    );
  }

  const [activeImageIndex, setActiveImageIndex] = useState(0);
  const [selectedColor, setSelectedColor] = useState(
    selectedProduct.selectedDefaultColor || selectedProduct.colors[0]?.name || ''
  );
  const [quantity, setQuantity] = useState(1);
  const [pincode, setPincode] = useState('');
  const [pincodeStatus, setPincodeStatus] = useState<string | null>(null);

  const isFavorited = isInWishlist(selectedProduct.id);

  const checkDelivery = (e: React.FormEvent) => {
    e.preventDefault();
    if (/^\d{6}$/.test(pincode.trim())) {
      setPincodeStatus('Available for Insured Speed Post (3-6 business days across India).');
    } else {
      setPincodeStatus('Please enter a valid 6-digit Indian PIN code.');
    }
  };

  const handleBuyNow = () => {
    addToCart(selectedProduct, quantity, selectedColor);
    navigate('checkout');
  };

  return (
    <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-10">
      {/* Breadcrumb */}
      <nav className="flex items-center gap-2 text-xs text-neutral-500 mb-8 uppercase tracking-wider">
        <button onClick={() => navigate('home')} className="hover:text-neutral-900">
          Home
        </button>
        <ChevronRight className="w-3.5 h-3.5" />
        <button onClick={() => navigate('shop')} className="hover:text-neutral-900">
          Catalog
        </button>
        <ChevronRight className="w-3.5 h-3.5" />
        <span className="text-neutral-900 font-semibold">{selectedProduct.brand}</span>
      </nav>

      {/* Product Hero */}
      <div className="grid grid-cols-1 lg:grid-cols-12 gap-12">
        {/* Images */}
        <div className="lg:col-span-7 space-y-4">
          <div className="aspect-square rounded-2xl overflow-hidden bg-white border border-neutral-200 shadow-md">
            <img
              src={selectedProduct.images[activeImageIndex] || selectedProduct.images[0]}
              alt={selectedProduct.name}
              className="w-full h-full object-cover"
            />
          </div>

          <div className="grid grid-cols-4 gap-3">
            {selectedProduct.images.map((img, idx) => (
              <button
                key={idx}
                onClick={() => setActiveImageIndex(idx)}
                className={`aspect-square rounded-lg overflow-hidden border-2 transition-all ${
                  activeImageIndex === idx ? 'border-amber-500 ring-2 ring-amber-500/20' : 'border-neutral-200 opacity-70 hover:opacity-100'
                }`}
              >
                <img src={img} alt="Thumbnail" className="w-full h-full object-cover" />
              </button>
            ))}
          </div>
        </div>

        {/* Buy Box */}
        <div className="lg:col-span-5 space-y-6">
          <div>
            <span className="text-xs font-bold uppercase tracking-widest text-amber-600">
              {selectedProduct.brand} • Ref: {selectedProduct.reference}
            </span>
            <h1 className="font-serif-display text-3xl sm:text-4xl font-bold text-neutral-950 mt-1">
              {selectedProduct.name}
            </h1>

            <div className="flex items-center gap-3 mt-2 text-sm text-neutral-600">
              <div className="flex items-center gap-1 text-amber-500 font-semibold">
                <Star className="w-4 h-4 fill-amber-400" />
                <span>{selectedProduct.rating}</span>
              </div>
              <span>•</span>
              <span>{selectedProduct.reviewsCount} Verified Customer Reviews</span>
            </div>
          </div>

          {/* Pricing */}
          <div className="p-4 bg-white rounded-xl border border-neutral-200">
            <div className="flex items-baseline gap-3">
              <span className="text-3xl font-extrabold text-neutral-950">
                ₹{selectedProduct.price.toLocaleString('en-IN')}
              </span>
              {selectedProduct.originalPrice > selectedProduct.price && (
                <span className="text-base text-neutral-400 line-through">
                  ₹{selectedProduct.originalPrice.toLocaleString('en-IN')}
                </span>
              )}
              {selectedProduct.discountPercent > 0 && (
                <span className="bg-emerald-100 text-emerald-800 text-xs font-bold px-2 py-0.5 rounded">
                  {selectedProduct.discountPercent}% OFF
                </span>
              )}
            </div>
            <p className="text-xs text-neutral-500 mt-1">
              Inclusive of all taxes. Free Insured Pan-India Speed Post.
            </p>
          </div>

          {/* Dial Selection */}
          {selectedProduct.colors.length > 0 && (
            <div>
              <label className="text-xs font-semibold uppercase tracking-wider text-neutral-700 block mb-2">
                Dial Colorway: <span className="text-neutral-900 font-bold">{selectedColor}</span>
              </label>
              <div className="flex items-center gap-3">
                {selectedProduct.colors.map((c) => (
                  <button
                    key={c.name}
                    onClick={() => setSelectedColor(c.name)}
                    className={`flex items-center gap-2 px-3 py-1.5 rounded-lg border text-xs font-medium transition-all ${
                      selectedColor === c.name
                        ? 'border-amber-600 bg-amber-50 text-neutral-950 font-bold'
                        : 'border-neutral-200 bg-white text-neutral-700 hover:bg-neutral-50'
                    }`}
                  >
                    <span
                      className="w-3.5 h-3.5 rounded-full border border-neutral-300"
                      style={{ backgroundColor: c.hex }}
                    />
                    <span>{c.name}</span>
                  </button>
                ))}
              </div>
            </div>
          )}

          {/* Quantity & Actions */}
          <div className="space-y-3 pt-2">
            <div className="flex items-center gap-4">
              <div className="flex items-center border border-neutral-300 rounded-lg bg-white">
                <button
                  onClick={() => setQuantity((q) => Math.max(1, q - 1))}
                  className="px-3 py-2 text-neutral-600 hover:bg-neutral-100"
                >
                  -
                </button>
                <span className="px-4 py-2 font-bold text-sm text-neutral-900">{quantity}</span>
                <button
                  onClick={() => setQuantity((q) => q + 1)}
                  className="px-3 py-2 text-neutral-600 hover:bg-neutral-100"
                >
                  +
                </button>
              </div>

              <button
                onClick={() => addToCart(selectedProduct, quantity, selectedColor)}
                className="flex-1 bg-neutral-950 hover:bg-neutral-900 text-white py-3 px-6 rounded-lg font-bold text-xs uppercase tracking-wider flex items-center justify-center gap-2 transition-colors shadow-md"
              >
                <ShoppingBag className="w-4 h-4" />
                Add to Bag
              </button>

              <button
                onClick={() => toggleWishlist(selectedProduct.id)}
                className={`p-3 border rounded-lg transition-colors ${
                  isFavorited ? 'border-rose-500 bg-rose-50 text-rose-600' : 'border-neutral-300 hover:bg-neutral-50'
                }`}
                title="Wishlist"
              >
                <Heart className={`w-5 h-5 ${isFavorited ? 'fill-current' : ''}`} />
              </button>
            </div>

            <button
              onClick={handleBuyNow}
              className="w-full bg-gradient-to-r from-amber-400 via-amber-500 to-amber-600 hover:from-amber-300 hover:to-amber-500 text-neutral-950 font-bold py-3.5 rounded-lg text-xs uppercase tracking-wider transition-all shadow-md"
            >
              Buy Now with Express Checkout
            </button>
          </div>

          {/* Pincode Checker */}
          <div className="bg-neutral-50 p-4 rounded-xl border border-neutral-200">
            <form onSubmit={checkDelivery} className="flex gap-2">
              <div className="relative flex-1">
                <MapPin className="w-4 h-4 text-neutral-400 absolute left-3 top-1/2 -translate-y-1/2" />
                <input
                  type="text"
                  maxLength={6}
                  value={pincode}
                  onChange={(e) => setPincode(e.target.value)}
                  placeholder="Enter 6-digit Indian PIN code"
                  className="w-full bg-white border border-neutral-300 rounded-lg pl-9 pr-3 py-2 text-xs focus:outline-none focus:border-amber-500"
                />
              </div>
              <button
                type="submit"
                className="bg-neutral-900 text-white text-xs font-semibold px-4 py-2 rounded-lg hover:bg-neutral-800"
              >
                Check
              </button>
            </form>
            {pincodeStatus && (
              <p className="mt-2 text-xs font-medium text-amber-900 flex items-center gap-1.5">
                <CheckCircle2 className="w-3.5 h-3.5 text-emerald-600" />
                {pincodeStatus}
              </p>
            )}
          </div>

          {/* Key Specs Matrix */}
          <div className="border-t border-neutral-200 pt-6 space-y-2 text-xs text-neutral-600">
            <div className="flex justify-between py-1.5 border-b border-neutral-100">
              <span className="text-neutral-500">Movement Caliber:</span>
              <span className="font-semibold text-neutral-900">{selectedProduct.movement}</span>
            </div>
            <div className="flex justify-between py-1.5 border-b border-neutral-100">
              <span className="text-neutral-500">Case Dimensions:</span>
              <span className="font-semibold text-neutral-900">
                {selectedProduct.caseSize} Diameter • {selectedProduct.thickness} Thickness
              </span>
            </div>
            <div className="flex justify-between py-1.5 border-b border-neutral-100">
              <span className="text-neutral-500">Water Resistance:</span>
              <span className="font-semibold text-neutral-900">{selectedProduct.waterResistance}</span>
            </div>
            <div className="flex justify-between py-1.5 border-b border-neutral-100">
              <span className="text-neutral-500">Crystal Glass:</span>
              <span className="font-semibold text-neutral-900">{selectedProduct.glassType}</span>
            </div>
            <div className="flex justify-between py-1.5">
              <span className="text-neutral-500">Warranty:</span>
              <span className="font-semibold text-neutral-900">{selectedProduct.warranty}</span>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
};
