import React, { useState, useMemo } from 'react';
import { useStore } from '../context/StoreContext';
import { ProductCard } from './ProductCard';
import { GenderCategory, WatchStyle } from '../types/store';
import { SlidersHorizontal, ArrowUpDown, X, Search } from 'lucide-react';

interface ShopCatalogProps {
  genderFilter?: GenderCategory;
  categoryTitle?: string;
  categorySubtitle?: string;
}

export const ShopCatalog: React.FC<ShopCatalogProps> = ({
  genderFilter,
  categoryTitle = 'N Collection Timepiece Catalog',
  categorySubtitle = 'Discover meticulously engineered watches calibrated for precision, durability, and distinguished style.'
}) => {
  const { products, searchFilterQuery, setSearchFilterQuery } = useStore();

  const [selectedGender, setSelectedGender] = useState<GenderCategory | 'All'>(
    genderFilter || 'All'
  );
  const [selectedStyle, setSelectedStyle] = useState<WatchStyle | 'All'>('All');
  const [selectedBrand, setSelectedBrand] = useState<string>('All');
  const [priceRange, setPriceRange] = useState<number>(100000);
  const [inStockOnly, setInStockOnly] = useState(false);
  const [sortBy, setSortBy] = useState<'featured' | 'price-asc' | 'price-desc' | 'rating'>(
    'featured'
  );
  const [mobileFilterOpen, setMobileFilterOpen] = useState(false);

  const brands = useMemo(() => {
    const list = Array.from(new Set(products.map((p) => p.brand)));
    return ['All', ...list];
  }, [products]);

  const filteredProducts = useMemo(() => {
    return products
      .filter((p) => {
        if (selectedGender !== 'All' && p.gender !== selectedGender) return false;
        if (selectedStyle !== 'All' && p.style !== selectedStyle) return false;
        if (selectedBrand !== 'All' && p.brand !== selectedBrand) return false;
        if (p.price > priceRange) return false;
        if (inStockOnly && (!p.isAvailable || p.stockCount <= 0)) return false;

        if (searchFilterQuery.trim()) {
          const q = searchFilterQuery.toLowerCase();
          const matches =
            p.name.toLowerCase().includes(q) ||
            p.brand.toLowerCase().includes(q) ||
            p.reference.toLowerCase().includes(q) ||
            p.category.toLowerCase().includes(q) ||
            p.style.toLowerCase().includes(q);
          if (!matches) return false;
        }

        return true;
      })
      .sort((a, b) => {
        if (sortBy === 'price-asc') return a.price - b.price;
        if (sortBy === 'price-desc') return b.price - a.price;
        if (sortBy === 'rating') return b.rating - a.rating;
        return (b.isBestseller ? 1 : 0) - (a.isBestseller ? 1 : 0);
      });
  }, [
    products,
    selectedGender,
    selectedStyle,
    selectedBrand,
    priceRange,
    inStockOnly,
    searchFilterQuery,
    sortBy
  ]);

  return (
    <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-10">
      {/* Header */}
      <div className="mb-8 border-b border-neutral-200 pb-6">
        <h1 className="font-serif-display text-3xl sm:text-5xl font-bold text-neutral-900">
          {categoryTitle}
        </h1>
        <p className="mt-2 text-neutral-600 max-w-3xl text-sm sm:text-base leading-relaxed">
          {categorySubtitle}
        </p>

        {/* Real-time search bar */}
        <div className="mt-6 max-w-xl relative">
          <Search className="w-4 h-4 absolute left-3.5 top-1/2 -translate-y-1/2 text-neutral-400" />
          <input
            type="text"
            value={searchFilterQuery}
            onChange={(e) => setSearchFilterQuery(e.target.value)}
            placeholder="Filter watches by brand, model name, movement..."
            className="w-full bg-white border border-neutral-300 text-neutral-900 pl-10 pr-10 py-2.5 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-amber-500/50 focus:border-amber-500"
          />
          {searchFilterQuery && (
            <button
              onClick={() => setSearchFilterQuery('')}
              className="absolute right-3 top-1/2 -translate-y-1/2 text-neutral-400 hover:text-neutral-700"
            >
              <X className="w-4 h-4" />
            </button>
          )}
        </div>
      </div>

      {/* Main Grid: Sidebar Filters + Products */}
      <div className="grid grid-cols-1 lg:grid-cols-4 gap-8">
        {/* Desktop Sidebar Filters */}
        <aside className="hidden lg:block space-y-6 bg-white p-6 rounded-xl border border-neutral-200 shadow-sm self-start sticky top-28">
          <div className="flex items-center justify-between border-b border-neutral-100 pb-4">
            <h3 className="font-serif-display text-lg font-bold text-neutral-900 flex items-center gap-2">
              <SlidersHorizontal className="w-4 h-4 text-amber-600" />
              Filter Collection
            </h3>
            <button
              onClick={() => {
                setSelectedGender('All');
                setSelectedStyle('All');
                setSelectedBrand('All');
                setPriceRange(100000);
                setInStockOnly(false);
                setSearchFilterQuery('');
              }}
              className="text-xs text-amber-700 hover:text-amber-800 font-medium"
            >
              Reset
            </button>
          </div>

          {/* Gender */}
          <div>
            <label className="text-xs uppercase tracking-wider font-semibold text-neutral-600 block mb-2">
              Gender Category
            </label>
            <div className="flex flex-wrap gap-2">
              {(['All', 'Men', 'Women', 'Unisex'] as const).map((g) => (
                <button
                  key={g}
                  onClick={() => setSelectedGender(g)}
                  className={`text-xs px-3 py-1.5 rounded-full border transition-colors ${
                    selectedGender === g
                      ? 'bg-neutral-950 text-white border-neutral-950'
                      : 'bg-neutral-50 text-neutral-700 border-neutral-200 hover:bg-neutral-100'
                  }`}
                >
                  {g}
                </button>
              ))}
            </div>
          </div>

          {/* Style */}
          <div>
            <label className="text-xs uppercase tracking-wider font-semibold text-neutral-600 block mb-2">
              Horological Style
            </label>
            <div className="flex flex-wrap gap-2">
              {(['All', 'Chronograph', 'Diver', 'Dress', 'Sport', 'Casual'] as const).map((s) => (
                <button
                  key={s}
                  onClick={() => setSelectedStyle(s as any)}
                  className={`text-xs px-3 py-1.5 rounded-full border transition-colors ${
                    selectedStyle === s
                      ? 'bg-neutral-950 text-white border-neutral-950'
                      : 'bg-neutral-50 text-neutral-700 border-neutral-200 hover:bg-neutral-100'
                  }`}
                >
                  {s}
                </button>
              ))}
            </div>
          </div>

          {/* Max Price Range Slider */}
          <div>
            <div className="flex items-center justify-between mb-2">
              <label className="text-xs uppercase tracking-wider font-semibold text-neutral-600">
                Max Price
              </label>
              <span className="text-xs font-bold text-neutral-900">
                ₹{priceRange.toLocaleString('en-IN')}
              </span>
            </div>
            <input
              type="range"
              min="10000"
              max="100000"
              step="5000"
              value={priceRange}
              onChange={(e) => setPriceRange(Number(e.target.value))}
              className="w-full accent-amber-500 cursor-pointer"
            />
          </div>

          {/* In Stock toggle */}
          <div className="pt-2 border-t border-neutral-100">
            <label className="flex items-center gap-2 cursor-pointer text-xs font-medium text-neutral-800">
              <input
                type="checkbox"
                checked={inStockOnly}
                onChange={(e) => setInStockOnly(e.target.checked)}
                className="w-4 h-4 rounded text-amber-600 focus:ring-amber-500"
              />
              In Stock & Ready to Ship Only
            </label>
          </div>
        </aside>

        {/* Product Cards Stream */}
        <div className="lg:col-span-3 space-y-6">
          {/* Top Sort Controls */}
          <div className="flex items-center justify-between bg-white px-4 py-3 rounded-lg border border-neutral-200">
            <span className="text-xs text-neutral-600 font-medium">
              Showing <strong className="text-neutral-900">{filteredProducts.length}</strong> timepieces
            </span>

            <div className="flex items-center gap-2">
              <ArrowUpDown className="w-3.5 h-3.5 text-neutral-500" />
              <select
                value={sortBy}
                onChange={(e) => setSortBy(e.target.value as any)}
                className="text-xs font-semibold bg-transparent border-none focus:outline-none cursor-pointer text-neutral-800"
              >
                <option value="featured">Featured First</option>
                <option value="price-asc">Price: Low to High</option>
                <option value="price-desc">Price: High to Low</option>
                <option value="rating">Highest Rated</option>
              </select>
            </div>
          </div>

          {/* Product Grid */}
          {filteredProducts.length > 0 ? (
            <div className="grid grid-cols-1 sm:grid-cols-2 xl:grid-cols-3 gap-6">
              {filteredProducts.map((product) => (
                <ProductCard key={product.id} product={product} />
              ))}
            </div>
          ) : (
            <div className="bg-white rounded-xl p-12 text-center border border-neutral-200">
              <p className="font-serif-display text-xl text-neutral-700">No timepieces matched your filters.</p>
              <p className="text-xs text-neutral-500 mt-2">Try clearing your search query or adjusting your price parameters.</p>
            </div>
          )}
        </div>
      </div>
    </div>
  );
};
